## Эталонный код на две кучи
---
```python
from functools import lru_cache


@lru_cache(None)
def f(a, b):
    if a+b >= 99: return 0 # лимит в куче
    steps = [f(a+2,b), f(a*3,b), f(a,b+2), f(a,b*3)] #просчет возможных ходов
    win = [i for i in steps if i <= 0] # отбор выигрышных позиций (враг в них будет проигрывать)
    if win:
        return -max(win)+1 #возвращаем самый быстрый к выйгрышу ход
    return -max(steps) #возвращаем самый медленный к проигрышу ход

for i in range(1, 94): # перебор ситуаций
    if f(5, i) == -2:
        print(i)
```
